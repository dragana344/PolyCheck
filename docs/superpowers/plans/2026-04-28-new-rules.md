# analyzer/languages/shared/consecutive_punctuation_rule.py
from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class ConsecutivePunctuationRule(BaseRule):
    """
    [HEURISTIC] Flags two or more consecutive punctuation characters.
    A valid ellipsis (...) is explicitly excluded.
    """

    name = "ConsecutivePunctuation"

    _PATTERN = re.compile(r'[!?.,;:]{2,}')

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        errors: List[DetectedError] = []
        for match in self._PATTERN.finditer(sentence):
            if match.group() == '...':
                continue
            errors.append(DetectedError(
                sentence=sentence,
                error_type="ConsecutivePunctuation",
                message=(
                    f"[HEURISTIC] Consecutive punctuation '{match.group()}' detected."
                ),
            ))
        return errors
```

# analyzer/languages/shared/unbalanced_quotes_rule.py
from __future__ import annotations

from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class UnbalancedQuotesRule(BaseRule):
    """
    [HEURISTIC] Flags a sentence that contains an odd number of double-quote
    characters, which typically indicates an unclosed or unopened quotation.
    Only double quotes are checked; single quotes are skipped because they
    overlap with apostrophes and produce too many false positives.
    """

    name = "UnbalancedQuotes"

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        if sentence.count('"') % 2 != 0:
            return [DetectedError(
                sentence=sentence,
                error_type="UnbalancedQuotes",
                message="[HEURISTIC] Odd number of double-quote characters — possible unclosed quotation.",
            )]
        return []
```

from __future__ import annotations

from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class UnbalancedParenthesesRule(BaseRule):
    """
    [HEURISTIC] Flags a sentence where the count of opening parentheses does
    not match the count of closing parentheses.
    """

    name = "UnbalancedParentheses"

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        if sentence.count('(') != sentence.count(')'):

                message="[HEURISTIC] Mismatched parentheses — opening and closing counts differ.",
            )]
        return []

# analyzer/languages/english/rules/article_before_vowel_rule.py
from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class ArticleBeforeVowelRule(BaseRule):
    """
    [HEURISTIC] Flags the indefinite article 'a' appearing directly before a
    vowel-starting word (should be 'an').  Note: words with a vowel letter but
    a consonant sound (e.g. 'unique', 'uniform') will be false positives.
    """

    name = "ArticleBeforeVowel"

    _PATTERN = re.compile(r'\ba\s+[aeiouAEIOU]', re.IGNORECASE)

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        errors: List[DetectedError] = []
        for match in self._PATTERN.finditer(sentence):
            errors.append(DetectedError(
                sentence=sentence,
                error_type="ArticleBeforeVowel",
                message=(
                    f"[HEURISTIC] '{match.group().strip()}' — use 'an' before a vowel sound."
                ),
            ))
        return errors
```

# analyzer/languages/english/rules/common_confusion_rule.py
from __future__ import annotations

import re
from typing import List, Tuple

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class CommonConfusionRule(BaseRule):
    """
    [HEURISTIC] Flags a hardcoded set of commonly confused word sequences.
    Each entry is a (pattern, wrong_form, suggestion) tuple.
    """

    name = "CommonConfusion"

    _CONFUSIONS: List[Tuple[re.Pattern, str, str]] = [
        (re.compile(r'\bshould\s+of\b', re.IGNORECASE), "should of", "should have"),
        (re.compile(r'\bwould\s+of\b', re.IGNORECASE), "would of", "would have"),
        (re.compile(r'\bcould\s+of\b', re.IGNORECASE), "could of", "could have"),
        (re.compile(r'\bmust\s+of\b', re.IGNORECASE), "must of", "must have"),
        (re.compile(r'\byour\s+welcome\b', re.IGNORECASE), "your welcome", "you're welcome"),
        (re.compile(r'\btheir\s+(is|are|was|were)\b', re.IGNORECASE), "their is/are/was/were", "there"),
    ]

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        errors: List[DetectedError] = []
        for pattern, wrong, suggestion in self._CONFUSIONS:
            if pattern.search(sentence):
                errors.append(DetectedError(
                    sentence=sentence,
                    error_type="CommonConfusion",
                    message=(
                        f"[HEURISTIC] '{wrong}' is commonly confused — "
                        f"did you mean '{suggestion}'?"
                    ),
                ))
        return errors
```

# analyzer/languages/english/rules/apostrophe_rule.py
from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class ApostropheRule(BaseRule):
    """
    [HEURISTIC] Flags common English contractions written without an apostrophe
    (e.g. 'dont' instead of 'don't').  The check is case-insensitive and
    uses whole-word matching to avoid false positives inside longer words.
    """

    name = "MissingApostrophe"

    _PATTERN = re.compile(
        r'\b(dont|doesnt|cant|wont|isnt|arent|wasnt|werent|havent|hasnt|'
        r'hadnt|wouldnt|couldnt|shouldnt|mustnt|theyre|youre|hes|shes|'
        r'weve|theyve|ive|youve)\b',
        re.IGNORECASE,
    )

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        errors: List[DetectedError] = []
        for match in self._PATTERN.finditer(sentence):
            errors.append(DetectedError(
                sentence=sentence,
                error_type="MissingApostrophe",
                message=(
                    f"[HEURISTIC] '{match.group()}' looks like a contraction "
                    f"missing an apostrophe."
                ),
            ))
        return errors
```

# analyzer/languages/english/rules/double_negation_rule.py
from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class DoubleNegationRule(BaseRule):
    """
    [HEURISTIC] Flags sentences that contain a negative auxiliary verb or
    'never' alongside a second negative word (no/nobody/nothing/nowhere/
    neither).  This catches the classic double-negative pattern
    ('I don't have no money') without requiring a full parse tree.
    """

    name = "DoubleNegation"

    _NEG_AUX = re.compile(
        r"\b(don't|doesn't|can't|won't|isn't|aren't|wasn't|weren't|"
        r"haven't|hasn't|hadn't|wouldn't|couldn't|shouldn't|never)\b",
        re.IGNORECASE,
    )
    _NEG_WORD = re.compile(
        r'\b(no|nobody|nothing|nowhere|neither)\b',
        re.IGNORECASE,
    )

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        if self._NEG_AUX.search(sentence) and self._NEG_WORD.search(sentence):
            return [DetectedError(
                sentence=sentence,
                error_type="DoubleNegation",
                message="[HEURISTIC] Possible double negation detected.",
            )]
        return []
```

from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError

_INTRODUCTORY = (
    'However', 'Therefore', 'Furthermore', 'Moreover', 'Nevertheless',
    'Consequently', 'Additionally', 'Finally', 'Meanwhile', 'Otherwise',
    'Instead', 'Indeed', 'Thus', 'Hence', 'Nonetheless', 'Subsequently',
    'First', 'Second', 'Third', 'Lastly',

class CommaAfterIntroductoryPhraseRule(BaseRule):
    """
    [HEURISTIC] Flags sentences that start with a common introductory word
    not followed by a comma (e.g. 'However the dog' should be
    'However, the dog').
    """

    name = "MissingIntroductoryComma"

    # Matches "Word " (space, no comma) at sentence start — case-insensitive.
    # "Word, " would not match because the comma breaks the \s+ sequence.
    _PATTERN = re.compile(
        r'^(' + '|'.join(_INTRODUCTORY) + r')\s+[A-Za-z]',
        re.IGNORECASE,
    )

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        match = self._PATTERN.match(sentence)
        if match:
            word = match.group(1)
            return [DetectedError(
                sentence=sentence,
                error_type="MissingIntroductoryComma",
                message=(
                    f"[HEURISTIC] '{word}' at the start of a sentence should "
                    f"be followed by a comma."
                ),
            )]
        return []

from __future__ import annotations

import re
from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError


class CyrillicVowelRule(BaseRule):
    """
    [HEURISTIC] Flags Cyrillic tokens of 2+ characters that contain no
    Macedonian vowel (а, е, и, о, у).  A real Macedonian word always contains
    at least one vowel; a vowel-free token is almost certainly a typo or
    garbled input.
    """

    name = "CyrillicNoVowel"

    _TOKEN_RE = re.compile(r'[Ѐ-ӿ]{2,}')
    _VOWELS = frozenset('аеиоуАЕИОУ')

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        errors: List[DetectedError] = []
        for match in self._TOKEN_RE.finditer(sentence):
            word = match.group()
            if not any(ch in self._VOWELS for ch in word):
                errors.append(DetectedError(
                    sentence=sentence,
                    error_type="CyrillicNoVowel",
                    message=(
                        f"[HEURISTIC] Кирилскиот збор '{word}' не содржи "
                        f"самогласка — можна печатна грешка."
                    ),
                ))
        return errors
```

# analyzer/languages/macedonian/rules/homoglyph_rule.py
from __future__ import annotations

from typing import List

from analyzer.base_rule import BaseRule
from analyzer.models import DetectedError

# Latin characters that are visually identical (or nearly so) to Cyrillic
# equivalents: a→а, e→е, o→о, c→с, p→р, x→х, A→А, B→В, C→С, E→Е,
# K→К, M→М, O→О, P→Р, T→Т, X→Х
_HOMOGLYPHS = frozenset('aeopcxABCEKMOPTX')


class HomoglyphRule(BaseRule):
    """
    [HEURISTIC] Detects Latin characters that are visually indistinguishable
    from Cyrillic characters in sentences that contain Cyrillic text.  These
    are invisible errors — the word looks correct on screen but contains the
    wrong Unicode codepoint (e.g. Latin 'a' instead of Cyrillic 'а').
    Only fires when at least one Cyrillic character is also present.
    """

    name = "HomoglyphDetection"

    def check(self, sentence: str, tokens: List[str]) -> List[DetectedError]:
        cyrillic_present = any('Ѐ' <= ch <= 'ӿ' for ch in sentence)
        if not cyrillic_present:
            return []

        found = sorted(set(ch for ch in sentence if ch in _HOMOGLYPHS))
        if not found:
            return []

        return [DetectedError(
            sentence=sentence,
            error_type="Homoglyph",
            message=(
                f"[HEURISTIC] Латински знаци {found} кои визуелно изгледаат "
                f"идентично со кирилски се пронајдени во кирилски текст — "
                f"можна ненамерна замена на знаци."
            ),
        )]
```
